class Power(ctypes.Structure):
    _fields_ = [('ac', ctypes.c_ubyte), ('flags', ctypes.c_ubyte),
        ('percent', ctypes.c_ubyte), ('saver', ctypes.c_ubyte),
        ('life', ctypes.c_uint32), ('full_life', ctypes.c_uint32)]

def power():
    status = Power()
    observed = dict(observedAtUtc=time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
        source='GetSystemPowerStatus', acConnected=None, batterySaver=None, chargePercent=None)
    if ctypes.windll.kernel32.GetSystemPowerStatus(ctypes.byref(status)):
        observed.update(acConnected={0: False, 1: True}.get(status.ac),
            batterySaver={0: False, 1: True}.get(status.saver),
            chargePercent=status.percent if status.percent <= 100 else None,
            batteryFlags=int(status.flags))
    return observed

def elapsed():

def save():

def run(name, command, target=None, tests=False, cwd=REPO):
    if PLAN['workCutoffSeconds'] - elapsed() < 5:
        raise RuntimeError('Original work cutoff reached; remaining checks stay pending.')
    command = [str(arg) for arg in command]
    check = dict(name=name, command=command, startedSeconds=elapsed(), powerBefore=power(), passed=False)
    receipt['checks'].append(check)
    save()
    print('Running ' + name, flush=True)
    childenv = env.copy()
    if target:
        childenv['CARGO_TARGET_DIR'] = str(target)
    with (TRIAL / (name + '.stdout.log')).open('wb') as out, (TRIAL / (name + '.stderr.log')).open('wb') as err:
        child = subprocess.Popen(command, cwd=cwd, env=childenv, stdout=out, stderr=err,
            creationflags=subprocess.CREATE_NO_WINDOW)
        try:
            check['exitCode'] = child.wait(timeout=max(1, PLAN['workCutoffSeconds'] - elapsed()))
            check['passed'] = check['exitCode'] == 0
        except subprocess.TimeoutExpired:
            check['failure'] = 'original-work-cutoff'
            cleanup = dict(treeKillExitCode=None, rootReaped=False, treeCleanupConfirmed=False)
            try:
                killed = subprocess.run([r'C:\Windows\System32\taskkill.exe', '/PID', str(child.pid), '/T', '/F'],
                    stdout=out, stderr=err, timeout=10, check=False, creationflags=subprocess.CREATE_NO_WINDOW)
                cleanup['treeKillExitCode'] = killed.returncode
            except (OSError, subprocess.TimeoutExpired):
                cleanup['treeKillObservation'] = 'unavailable'
            finally:
                try:
                    if child.poll() is None:
                        child.kill()
                    child.wait(timeout=5)
                    cleanup['rootReaped'] = True
                except (OSError, subprocess.TimeoutExpired):
                    cleanup['rootReapObservation'] = 'unavailable'
                cleanup['treeCleanupConfirmed'] = cleanup['treeKillExitCode'] == 0 and cleanup['rootReaped']
                check['timeoutCleanup'] = cleanup
        finally:
            check.update(finishedSeconds=elapsed(), powerAfter=power())
    stdout = (TRIAL / (name + '.stdout.log')).read_text(encoding='utf-8', errors='replace')
    if tests and check['passed']:
        check['testsPassed'] = sum(map(int, re.findall(r'test result: ok\. (\d+) passed;', stdout)))
        check['passed'] = check['testsPassed'] > 0
    save()
    if not check['passed']:
        print((TRIAL / (name + '.stderr.log')).read_text(encoding='utf-8', errors='replace')[-9000:], flush=True)
        print(stdout[-5000:], flush=True)
        raise RuntimeError('Check failed: ' + name)
    print(name + ' passed' + (': ' + str(check['testsPassed']) + ' tests' if tests else ''), flush=True)
    return stdout
