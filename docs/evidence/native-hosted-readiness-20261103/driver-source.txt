class Power(ctypes.Structure):
    _fields_ = [('ac', ctypes.c_ubyte), ('flags', ctypes.c_ubyte),
        ('percent', ctypes.c_ubyte), ('saver', ctypes.c_ubyte),
        ('life', ctypes.c_uint32), ('full_life', ctypes.c_uint32)]

def power():
    status = Power()
    result = dict(observedAtUtc=time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
        source='GetSystemPowerStatus', acConnected=None, batterySaver=None, chargePercent=None)
    if ctypes.windll.kernel32.GetSystemPowerStatus(ctypes.byref(status)):
        result.update(acConnected={0: False, 1: True}.get(status.ac),
            batterySaver={0: False, 1: True}.get(status.saver),
            chargePercent=status.percent if status.percent <= 100 else None,
            batteryFlags=int(status.flags))
    return result

started = time.monotonic()
receipt = dict(schemaVersion=1, kind=PLAN['kind'],
    startedAtUtc=time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
    maximumTotalSeconds=PLAN['maximumTotalSeconds'], workCutoffSeconds=PLAN['workCutoffSeconds'],

def elapsed():
    return time.monotonic() - started

def run(name, arguments):
    remaining = PLAN['workCutoffSeconds'] - elapsed()
    if remaining < 5:
        raise RuntimeError('Original work cutoff reached; check remains pending.')
    check = dict(name=name, command=[str(arg) for arg in arguments], startedSeconds=elapsed(),
        powerBefore=power(), passed=False)
    receipt['checks'].append(check)
    with (TRIAL / (name + '.stdout.log')).open('wb') as out, (TRIAL / (name + '.stderr.log')).open('wb') as err:
        child = subprocess.Popen(check['command'], cwd=REPO, stdout=out, stderr=err,
            creationflags=subprocess.CREATE_NO_WINDOW)
        try:
            check['exitCode'] = child.wait(timeout=max(1, PLAN['workCutoffSeconds'] - elapsed()))
            check['passed'] = check['exitCode'] == 0
        except subprocess.TimeoutExpired:
            check['failure'] = 'original-work-cutoff'
            cleanup = dict(treeKillExitCode=None, rootReaped=False, treeCleanupConfirmed=False)
            try:
                result = subprocess.run([str(Path(os.environ['SystemRoot']) / 'System32/taskkill.exe'),
                    '/PID', str(child.pid), '/T', '/F'], stdout=out, stderr=err,
                    timeout=10, check=False, creationflags=subprocess.CREATE_NO_WINDOW)
                cleanup['treeKillExitCode'] = result.returncode
            except (OSError, subprocess.TimeoutExpired):
                cleanup['treeKillObservation'] = 'unknown'
            finally:
                try:
                    if child.poll() is None:
                        child.kill()
                    child.wait(timeout=5)
                    cleanup['rootReaped'] = True
                except (OSError, subprocess.TimeoutExpired):
                    cleanup['rootReapObservation'] = 'unknown'
                cleanup['treeCleanupConfirmed'] = cleanup['treeKillExitCode'] == 0 and cleanup['rootReaped']
                check['timeoutCleanup'] = cleanup
        finally:
            check.update(finishedSeconds=elapsed(), powerAfter=power())
    if not check['passed']:
        raise RuntimeError('Check failed: ' + name)
    return (TRIAL / (name + '.stdout.log')).read_text(encoding='utf-8-sig')

def ps_string(value):
    return "'" + str(value).replace("'", "''") + "'"

}
'''

foreach ($guard in @('Test-Publication', 'Test-Intake')) {

foreach ($fixture in $fixtures.Outputs) {

[ordered]@{ GuardChecksPassed=$checks; ParsedFiles=$parseFiles.Count } | ConvertTo-Json -Compress
