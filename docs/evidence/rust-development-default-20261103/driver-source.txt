class Power(ctypes.Structure):
    _fields_ = [('ac', ctypes.c_ubyte), ('flags', ctypes.c_ubyte),
        ('percent', ctypes.c_ubyte), ('saver', ctypes.c_ubyte),
        ('life', ctypes.c_uint32), ('full_life', ctypes.c_uint32)]

def power():
    status = Power()
    result = dict(observedAtUtc=time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
        source='GetSystemPowerStatus', acConnected=None, batterySaver=None, chargePercent=None)
    if ctypes.windll.kernel32.GetSystemPowerStatus(ctypes.byref(status)):
        result.update(acConnected={0: False, 1: True}.get(status.ac),
            batterySaver={0: False, 1: True}.get(status.saver),
            chargePercent=status.percent if status.percent <= 100 else None,
            batteryFlags=int(status.flags))
    return result

def run(name, command, target=None, tests=False):
    remaining = 540 - elapsed()
    if remaining < 5:
        raise RuntimeError('Original work cutoff reached; remaining checks stay pending.')
    check = dict(name=name, command=[str(arg) for arg in command], startedSeconds=elapsed(), powerBefore=power(), passed=False)
    receipt['checks'].append(check)
    save()
    print('Running ' + name, flush=True)
    jobenv = env.copy()
    if target:
        jobenv['CARGO_TARGET_DIR'] = str(target)
    with (ROOT / (name + '.stdout.log')).open('wb') as out, (ROOT / (name + '.stderr.log')).open('wb') as err:
        child = subprocess.Popen(command, cwd=REPO, env=jobenv, stdout=out, stderr=err, creationflags=subprocess.CREATE_NO_WINDOW)
        try:
            check['exitCode'] = child.wait(timeout=max(1, 540 - elapsed()))
            check['passed'] = check['exitCode'] == 0
        except subprocess.TimeoutExpired:
            check['failure'] = 'original-work-cutoff'
            subprocess.run(['C:/Windows/System32/taskkill.exe', '/PID', str(child.pid), '/T', '/F'],
                stdout=out, stderr=err, timeout=10, check=False)
            child.wait(timeout=5)
        finally:
            check.update(finishedSeconds=elapsed(), powerAfter=power())
    if tests and check['passed']:
        output = (ROOT / (name + '.stdout.log')).read_text(encoding='utf-8', errors='replace')
        matches = re.findall(r'test result: ok\. (\d+) passed;', output)
        check['testsPassed'] = sum(map(int, matches))
        check['passed'] = check['testsPassed'] > 0
    save()
    if not check['passed']:
        print((ROOT / (name + '.stderr.log')).read_text(encoding='utf-8', errors='replace')[-7000:], flush=True)
        print((ROOT / (name + '.stdout.log')).read_text(encoding='utf-8', errors='replace')[-5000:], flush=True)
        raise RuntimeError('Check failed: ' + name)
    print(name + ' passed' + (': ' + str(check['testsPassed']) + ' tests' if tests else ''), flush=True)
